cmake_minimum_required(VERSION 3.20)
project(LxcHybrid LANGUAGES C)
find_package(Lua 5.4 EXACT REQUIRED)
add_library(lxc_hybrid MODULE runtime.c)
target_include_directories(lxc_hybrid PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(lxc_hybrid PRIVATE ${LUA_LIBRARIES})
set_target_properties(lxc_hybrid PROPERTIES PREFIX "" C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    set_property(TARGET lxc_hybrid PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(lxc_hybrid PRIVATE /W4 /WX)
else()
    target_compile_options(lxc_hybrid PRIVATE -Wall -Wextra -Werror)
endif()
