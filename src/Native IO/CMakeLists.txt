cmake_minimum_required(VERSION 3.20)
project(LuaExIO LANGUAGES CXX)
find_package(Lua 5.4 EXACT REQUIRED)
add_library(luaex_io MODULE Filesystem.cpp)
target_compile_features(luaex_io PRIVATE cxx_std_17)
target_include_directories(luaex_io PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(luaex_io PRIVATE ${LUA_LIBRARIES})
set_target_properties(luaex_io PROPERTIES PREFIX "")
if(MSVC)
    set_property(TARGET luaex_io PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(luaex_io PRIVATE /W4 /WX)
endif()
if(WIN32)
    set(LUAEX_SYSTEM windows)
elseif(APPLE)
    set(LUAEX_SYSTEM macos)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(LUAEX_SYSTEM linux)
else()
    message(FATAL_ERROR "Unsupported LuaEx I/O platform: ${CMAKE_SYSTEM_NAME}")
endif()
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" LUAEX_PROCESSOR)
if(WIN32 AND CMAKE_GENERATOR_PLATFORM)
    string(TOLOWER "${CMAKE_GENERATOR_PLATFORM}" LUAEX_GENERATOR_ARCH)
    if(LUAEX_GENERATOR_ARCH STREQUAL "arm64")
        set(LUAEX_ARCH arm64)
    elseif(LUAEX_GENERATOR_ARCH STREQUAL "x64")
        set(LUAEX_ARCH x64)
    else()
        message(FATAL_ERROR "Unsupported Windows target architecture: ${CMAKE_GENERATOR_PLATFORM}")
    endif()
elseif(LUAEX_PROCESSOR MATCHES "^(amd64|x86_64|x64)$")
    set(LUAEX_ARCH x64)
elseif(LUAEX_PROCESSOR MATCHES "^(arm64|aarch64)$")
    set(LUAEX_ARCH arm64)
else()
    message(FATAL_ERROR "Unsupported LuaEx I/O architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    if(CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        set(LUAEX_ARCH arm64)
    elseif(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
        set(LUAEX_ARCH x64)
    else()
        message(FATAL_ERROR "Build one macOS architecture at a time.")
    endif()
endif()
set_target_properties(luaex_io PROPERTIES OUTPUT_NAME "luaex_io-lua54-${LUAEX_SYSTEM}-${LUAEX_ARCH}")
install(TARGETS luaex_io LIBRARY DESTINATION "." RUNTIME DESTINATION ".")
