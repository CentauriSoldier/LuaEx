cmake_minimum_required(VERSION 3.20)
project(LxcNativePrototype LANGUAGES C)
find_package(Lua 5.4 EXACT REQUIRED)
add_library(lxc_native MODULE runtime.c)
target_include_directories(lxc_native PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(lxc_native PRIVATE ${LUA_LIBRARIES})
set_target_properties(lxc_native PROPERTIES PREFIX "" C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    set_property(TARGET lxc_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(lxc_native PRIVATE /W4 /WX)
else()
    target_compile_options(lxc_native PRIVATE -Wall -Wextra -Werror)
endif()
